struct StateHash {
    size_t operator()(const array<uint32_t,400>& state)const{
        uint64_t h=0x9e3779b97f4a7c15ULL;for(uint32_t v:state)h^=uint64_t(v)+0x9e3779b97f4a7c15ULL+(h<<6)+(h>>2);return size_t(h);
    }
};
using Visited=unordered_set<array<uint32_t,400>,StateHash>;

int choose(const Board& b,const BoardState& state,const Encoded& e,const FactoredActions& a,const Visited& visited,BoardState& next) {
    vector<int> towers;vector<float> pl;
    for(int p=0;p<b.cell_count;++p)if(a.tower[p].valid()){towers.push_back(p);pl.push_back(tower_logit(e,a,p));}
    if(towers.empty())return -1;
    const auto po=descending(pl);const auto pp=log_probabilities(pl);ConditionalActor actor(b,e);
    auto unseen=[&](int index){next=state;next.apply(b,a.moves[index]);if(visited.contains(next.bits)){++factor_counts.duplicates;return false;}return true;};
#ifdef V097_TOP2
    for(size_t start=0;start<po.size();start+=2){
        vector<pair<double,int>> candidates;
        for(size_t ti=start;ti<min(start+2,po.size());++ti){
            int slot=po[ti],p=towers[slot];vector<int> ks;vector<float> kl;
            for(int k=0;k<8;++k)if(a.split[p][k].valid()){ks.push_back(k);kl.push_back(split_logit(e,a,p,k));}
            auto kp=log_probabilities(kl);
            for(size_t ki=0;ki<ks.size();++ki){const auto& ids=a.groups[p][ks[ki]];vector<float> dl;
                for(int index:ids)dl.push_back(actor.logit(a.moves[index],a.features[index]));auto dp=log_probabilities(dl);
                for(size_t j=0;j<ids.size();++j)candidates.emplace_back(pp[slot]+kp[ki]+dp[j],ids[j]);}
        }
        stable_sort(candidates.begin(),candidates.end(),[](auto a,auto b){return a.first==b.first ? a.second<b.second : a.first>b.first;});
        for(auto [prob,index]:candidates){(void)prob;if(unseen(index))return index;}
    }
#else
    for(int slot:po){int p=towers[slot];vector<int> ks;vector<float> kl;
        for(int k=0;k<8;++k)if(a.split[p][k].valid()){ks.push_back(k);kl.push_back(split_logit(e,a,p,k));}
        for(int ki:descending(kl)){const auto& ids=a.groups[p][ks[ki]];vector<float> dl;
            for(int index:ids)dl.push_back(actor.logit(a.moves[index],a.features[index]));
            for(int j:descending(dl))if(unseen(ids[j]))return ids[j];}
    }
#endif
    return -1;
}
